# UIRecorder 前端设计文档

## 1. 技术栈

- **框架**: 原生 JavaScript (无框架)
- **样式**: CSS3 + Tailwind CSS
- **构建**: 纯静态 HTML
- **依赖**: 
  - Marked.js (Markdown渲染)
  - DOMPurify (XSS防护)
  - html2canvas (图片导出)

```javascript
  const reader = response.body.getReader();
  const decoder = new TextDecoder();
  let result = '';
  
  while (true) {
    const { done, value } = await reader.read();
    if (done) break;
    
    const chunk = decoder.decode(value);
    result += chunk;
    updateAIProgress(result);
  }
}
```

#### 撤销重做
```javascript
class HistoryManager {
  constructor() {
    this.undoStack = [];
    this.redoStack = [];
    this.maxSize = 50;
  }
  
  saveState(state) {
    this.undoStack.push(JSON.stringify(state));
    this.redoStack = [];
    if (this.undoStack.length > this.maxSize) {
      this.undoStack.shift();
    }
  }
  
  undo() {
    if (this.undoStack.length > 1) {
      this.redoStack.push(this.undoStack.pop());
      return JSON.parse(this.undoStack[this.undoStack.length - 1]);
    }
    return null;
  }
  
  redo() {
    if (this.redoStack.length > 0) {
      const state = this.redoStack.pop();
      this.undoStack.push(state);
      return JSON.parse(state);
    }
    return null;
  }
}
```

```javascript
function setupDraggable(element, handle) {
  let isDragging = false;
  let startX, startY;
  
  handle.addEventListener('mousedown', (e) => {
    isDragging = true;
    startX = e.clientX - element.offsetLeft;
    startY = e.clientY - element.offsetTop;
  });
  
  document.addEventListener('mousemove', (e) => {
    if (!isDragging) return;
    element.style.left = (e.clientX - startX) + 'px';
    element.style.top = (e.clientY - startY) + 'px';
  });
  
  document.addEventListener('mouseup', () => {
    isDragging = false;
  });
}
```

### 6.1 图片加载优化

```javascript
// 懒加载
function lazyLoadImages() {
  const images = document.querySelectorAll('img[data-src]');
  const observer = new IntersectionObserver((entries) => {
    entries.forEach(entry => {
      if (entry.isIntersecting) {
        const img = entry.target;
        img.src = img.dataset.src;
        observer.unobserve(img);
      }
    });
  });
  
  images.forEach(img => observer.observe(img));
}

// 缩略图生成
function generateThumbnail(imageUrl, maxSize = 200) {
  return new Promise((resolve) => {
    const img = new Image();
    img.crossOrigin = 'anonymous';
    img.onload = () => {
      const canvas = document.createElement('canvas');
      const scale = Math.min(maxSize / img.width, maxSize / img.height);
      canvas.width = img.width * scale;
      canvas.height = img.height * scale;
      const ctx = canvas.getContext('2d');
      ctx.drawImage(img, 0, 0, canvas.width, canvas.height);
      resolve(canvas.toDataURL('image/jpeg', 0.8));
    };
    img.src = imageUrl;
  });
}
```

### 6.2 虚拟滚动

```javascript
class VirtualList {
  constructor(container, itemHeight, renderFn) {
    this.container = container;
    this.itemHeight = itemHeight;
    this.renderFn = renderFn;
    this.visibleCount = Math.ceil(container.clientHeight / itemHeight);
    this.startIndex = 0;
    
    container.addEventListener('scroll', () => {
      this.updateVisibleItems();
    });
  }
  
  updateVisibleItems() {
    const scrollTop = this.container.scrollTop;
    this.startIndex = Math.floor(scrollTop / this.itemHeight);
    const endIndex = Math.min(this.startIndex + this.visibleCount, this.data.length);
    
    this.render(this.data.slice(this.startIndex, endIndex));
  }
}
```

## 7. API集成

### 7.1 数据获取

```javascript
class APIClient {
  constructor(baseURL) {
    this.baseURL = baseURL;
  }
  
  async get(endpoint) {
    const response = await fetch(`${this.baseURL}${endpoint}`);
    if (!response.ok) throw new Error('请求失败');
    return response.json();
  }
  
  async post(endpoint, data) {
    const response = await fetch(`${this.baseURL}${endpoint}`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(data)
    });
    if (!response.ok) throw new Error('请求失败');
    return response.json();
  }
}

const api = new APIClient('http://localhost:52000/api');
```

```javascript
class WebSocketClient {
```
